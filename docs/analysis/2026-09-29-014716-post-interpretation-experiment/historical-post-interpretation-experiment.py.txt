def encoded(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()


def digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()


def emit(kind, value):
    global _event_number
    _event_number += 1
    data = base64.b64encode(zlib.compress(encoded({"kind": kind, "value": value}))).decode()
    chunks = [data[n:n + 2400] for n in range(0, len(data), 2400)]
    for index, chunk in enumerate(chunks, 1):
        print(f"PW_INTERP {_event_number}:{index}/{len(chunks)}:{chunk}", flush=True)

def source_posts(cases):
    # Intentionally excludes expected judgments, tracked IDs, rubric and kinds.
    return {f"P{i:02d}": {"source_text": c["source_text"], "context": []}
            for i, c in enumerate(cases, 1)}


def validate_interpretation(value, sources):
    errors = []
    if not isinstance(value, dict) or set(value) != {"posts"} or not isinstance(value["posts"], dict) or set(value["posts"]) != set(sources):
        return ["post_keys_or_root_invalid"]
    expected_fields = {
        "entities": {"name", "roles", "evidence"},
        "claims": {"speaker", "subject", "statement", "evidence"},
        "calls_to_action": {"offering", "provider", "action", "evidence"},
    }
    for slot, row in value["posts"].items():
        if not isinstance(row, dict) or set(row) != {*expected_fields, "uncertainties"}:
            errors.append(f"{slot}:fields_invalid")
            continue
        for field, keys in expected_fields.items():
            items = row[field]
            if not isinstance(items, list):
                errors.append(f"{slot}:{field}:not_array")
                continue
            for item in items:
                if not isinstance(item, dict) or set(item) != keys:
                    errors.append(f"{slot}:{field}:item_shape")
                    continue
                for key in keys - {"roles"}:
                    if not isinstance(item[key], str) or not item[key].strip():
                        errors.append(f"{slot}:{field}:{key}:missing")
                evidence = item.get("evidence")
                if not isinstance(evidence, str) or not evidence or evidence not in sources[slot]["source_text"]:
                    errors.append(f"{slot}:{field}:evidence_not_verbatim")
                if field == "entities" and (not isinstance(item["roles"], list) or not item["roles"] or any(not isinstance(r, str) or not r.strip() for r in item["roles"])):
                    errors.append(f"{slot}:entity_roles_invalid")
        if not isinstance(row["uncertainties"], list) or any(not isinstance(u, str) for u in row["uncertainties"]):
            errors.append(f"{slot}:uncertainty_invalid")
    return errors


class RecordedTransport:
    """Capture the raw provider envelope before the production parser sees it."""
    def __init__(self, delegate, *, call_limit=MAX_CALLS, cost_limit=MAX_COST, emit_fn=emit):
        self.delegate = delegate
        self.call_limit = call_limit
        self.cost_limit = Decimal(cost_limit)
        self.emit_fn = emit_fn
        self.calls = 0
        self.cost = Decimal(0)
        self.input_tokens = 0
        self.output_tokens = 0
        self.blocked = None
        self.phase = ""
        self.records = []

    def __call__(self, request, timeout):
        bound = Decimal(len(encoded(request)) + 2048 + request["max_tokens"]) * Decimal("0.00002")
        if self.blocked or self.calls >= self.call_limit or self.cost + bound > self.cost_limit:
            raise DeepInfraPermanentError(self.blocked or "experiment_budget_reached")
        if request["model"] != MODEL:
            raise DeepInfraPermanentError("experiment_model_drift")
        self.calls += 1
        started = time.monotonic()
        record = {"call": self.calls, "phase": self.phase, "request": request,
                  "request_sha256": digest(request), "reserved_cost_usd": str(bound)}
        self.emit_fn("call_started", {k: v for k, v in record.items() if k != "request"})
        try:
            response = self.delegate._send_request(request, timeout=timeout)
            record["raw_response"] = response
            usage = response.get("usage") or {}
            raw_cost = usage.get("estimated_cost", usage.get("cost"))
            if raw_cost is None:
                raise ValueError("unknown_provider_cost")
            cost = Decimal(str(raw_cost))
            if not cost.is_finite() or cost < 0:
                raise ValueError("invalid_provider_cost")
            for key in ("prompt_tokens", "completion_tokens"):
                if not isinstance(usage.get(key), int) or usage[key] < 0:
                    raise ValueError("unknown_token_usage")
            self.cost += cost
            self.input_tokens += usage["prompt_tokens"]
            self.output_tokens += usage["completion_tokens"]
            if cost > bound or self.cost > self.cost_limit:
                raise ValueError("provider_cost_exceeds_bound")
            return response
        except Exception as exc:
            self.blocked = f"transport_or_usage_failure:{type(exc).__name__}"
            record["error"] = self.blocked
            # Never retry an uncertain billed call.
            raise DeepInfraPermanentError(self.blocked) from exc
        finally:
            record["latency_ms"] = round((time.monotonic() - started) * 1000)
            self.records.append(record)
            self.emit_fn("call_finished", record)


class InterpretationClient:
    def __init__(self, delegate, interpretations=None):
        self.delegate = delegate
        self.interpretations = interpretations

    def __getattr__(self, name):
        return getattr(self.delegate, name)

    def messages_create(self, **kwargs):
        if self.interpretations is not None:
            kwargs = deepcopy(kwargs)
            packet = json.loads(kwargs["messages"][0]["content"])
            if set(packet["cases"]) != set(self.interpretations):
                raise DeepInfraPermanentError("interpretation_slot_mismatch")
            packet["post_interpretations"] = self.interpretations
            kwargs["messages"][0]["content"] = json.dumps(packet, ensure_ascii=False, separators=(",", ":"))
            kwargs["system"] += "\n\n" + DERIVED_INSTRUCTION
        return self.delegate.messages_create(**kwargs)


def invoke_classifier(cases, registry, client):
    return classify_batch_pragmatics_full(
        [prior._packet(c) for c in cases], registry, client,
        model=MODEL, max_tokens=6144, max_workers=1, thinking={"type": "disabled"},
    )


def score(case, row):
    if "expected" in case:
        return prior._compare(case["expected"], row, case["tracked_brand_id"])
    checks = {}
    for brand, requirements in case["checks"].items():
        actual = row.get("by_brand", {}).get(brand, {})
        for key, value in requirements.items():
            if key.endswith("_contains"):
                field = key.removesuffix("_contains")
                okay = set(value).issubset(actual.get(field, []))
            elif key.endswith("_excludes"):
                field = key.removesuffix("_excludes")
                okay = not set(value).intersection(actual.get(field, []))
            else:
                okay = actual.get(key) == value
            checks[f"{brand}:{key}"] = bool(row.get("valid")) and okay
    return {"valid": bool(row.get("valid")), "checks": checks,
            "boundary_match": all(checks.values())}


def json_call(client, system, packet, max_tokens=6144):
    try:
        return {"value": dict(client.messages_create(
            model=MODEL, max_tokens=max_tokens, system=system,
            messages=[{"role": "user", "content": json.dumps(packet, ensure_ascii=False)}],
            temperature=1.0, top_p=1.0, seed=42, timeout=90,
        ))}
    except DeepInfraPermanentError as exc:
        return {"error": str(exc)}


def manifest():
    return {"experiment": EXPERIMENT, "model": MODEL, "profile": PROFILE,
            "prior_calls": 13, "planned_calls": 21, "maximum_new_calls": MAX_CALLS,
            "new_cost_cap_usd": str(MAX_COST), "batches": cohort(),
            "interpretation_prompt": INTERPRETATION_PROMPT,
            "derived_instruction": DERIVED_INSTRUCTION, "probe_prompt": PROBE_PROMPT,
            "script_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}


def run():
    frozen = manifest()
    emit("manifest", frozen)
    cfg = load_config(Path("config.yaml"))
    delegate = build_classifier_client_from_env(cfg)
    if not isinstance(delegate, DeepInfraChatCompletionsClient) or delegate.model != MODEL or delegate.request_profile != PROFILE:
        raise RuntimeError("configured_direct_classifier_unavailable_or_changed")
    recorded = RecordedTransport(delegate)
    client = replace(delegate, transport=recorded)
    batches = frozen["batches"]
    registry = prior._registry([c for batch in batches for c in batch])
    results = []
    for index, cases in enumerate(batches, 1):
        sources = source_posts(cases)
        result = {"batch": index, "case_ids": [c["case_id"] for c in cases]}
        # Alternate arm ordering between batches; extraction always precedes candidate.
        def baseline():
            recorded.phase = f"batch{index}:baseline_classifier"
            result["baseline"] = invoke_classifier(cases, registry, InterpretationClient(client))
        if index % 2:
            baseline()
        recorded.phase = f"batch{index}:interpretation"
        interpretation = json_call(client, INTERPRETATION_PROMPT, {"posts": sources})
        result["interpretation"] = interpretation
        errors = validate_interpretation(interpretation.get("value"), sources)
        result["interpretation_validation_errors"] = errors
        if not errors:
            derived = interpretation["value"]["posts"]
            recorded.phase = f"batch{index}:candidate_classifier"
            result["candidate"] = invoke_classifier(cases, registry, InterpretationClient(client, derived))
        else:
            derived = None
            result["candidate"] = [{"valid": False, "by_brand": {}, "error": "invalid_interpretation"} for _ in cases]
        if not index % 2:
            baseline()
        result["scores"] = [
            {"case_id": case["case_id"], "baseline": score(case, before), "candidate": score(case, after)}
            for case, before, after in zip(cases, result["baseline"], result["candidate"], strict=True)
        ]
        for arm in (("baseline", "candidate") if index % 2 else ("candidate", "baseline")):
            if arm == "candidate" and derived is None:
                result[f"probe_{arm}"] = {"error": "invalid_interpretation"}
                continue
            recorded.phase = f"batch{index}:{arm}_cross_task_probe"
            packet = {"posts": sources}
            system = PROBE_PROMPT
            if arm == "candidate":
                packet["post_interpretations"] = derived
                system += "\n\n" + DERIVED_INSTRUCTION
            result[f"probe_{arm}"] = json_call(client, system, packet)
        results.append(result)
        emit("batch_result", result)
        print(json.dumps({"batch_complete": index, "calls": recorded.calls,
                          "cost_usd": str(recorded.cost), "interpretation_errors": errors}), flush=True)
        if recorded.blocked:
            break
    summary = {"experiment": EXPERIMENT, "manifest_sha256": digest(frozen),
               "completed_batches": len(results), "calls": recorded.calls,
               "task_calls_total": 13 + recorded.calls, "cost_usd": str(recorded.cost),
               "input_tokens": recorded.input_tokens, "output_tokens": recorded.output_tokens,
               "blocked": recorded.blocked, "results": results}
    emit("final_result", summary)
    print(json.dumps({k: v for k, v in summary.items() if k != "results"}), flush=True)


def followup(path):
    """Diagnostic arm: reuse unchanged extractions with quote-only failures."""
    previous = json.loads(Path(path).read_text())
    if previous["blocked"] or previous["calls"] != 15:
        raise ValueError("unexpected_prior_budget_or_run")
    selected = [r for r in previous["results"] if r["interpretation_validation_errors"]]
    if len(selected) != 2 or any(
        not error.endswith(":evidence_not_verbatim")
        for row in selected for error in row["interpretation_validation_errors"]
    ):
        raise ValueError("followup_is_only_for_observed_quote_errors")
    emit("followup_contract", {"experiment": EXPERIMENT + "-quote-tolerant-diagnostic",
         "prior_experiment_calls": previous["calls"], "prior_task_calls": 28,
         "max_calls": 6, "prior_result_sha256": digest(previous),
         "scope": "Reuse the exact unedited model extractions for batches 1 and 2; bypass only verbatim quote admission. Original strict failures remain failures."})
    delegate = build_classifier_client_from_env(load_config(Path("config.yaml")))
    if not isinstance(delegate, DeepInfraChatCompletionsClient) or delegate.model != MODEL or delegate.request_profile != PROFILE:
        raise RuntimeError("configured_direct_classifier_unavailable_or_changed")
    recorded = RecordedTransport(delegate, call_limit=6)
    client = replace(delegate, transport=recorded)
    batches = cohort()
    registry = prior._registry([c for batch in batches for c in batch])
    results = []
    for previous_batch in selected:
        index = previous_batch["batch"]
        cases = batches[index - 1]
        derived = previous_batch["interpretation"]["value"]["posts"]
        recorded.phase = f"batch{index}:quote_tolerant_candidate_classifier"
        rows = invoke_classifier(cases, registry, InterpretationClient(client, derived))
        recorded.phase = f"batch{index}:quote_tolerant_candidate_cross_task_probe"
        probe = json_call(client, PROBE_PROMPT + "\n\n" + DERIVED_INSTRUCTION,
                          {"posts": source_posts(cases), "post_interpretations": derived})
        result = {"batch": index, "case_ids": previous_batch["case_ids"],
                  "admission_errors_retained": previous_batch["interpretation_validation_errors"],
                  "interpretation_sha256": digest(derived), "candidate": rows,
                  "scores": [{"case_id": c["case_id"], "candidate": score(c, r)}
                             for c, r in zip(cases, rows, strict=True)], "probe_candidate": probe}
        results.append(result)
        emit("batch_result", result)
        if recorded.blocked:
            break
    summary = {"experiment": EXPERIMENT + "-quote-tolerant-diagnostic", "calls": recorded.calls,
               "task_calls_total": 28 + recorded.calls, "cost_usd": str(recorded.cost),
               "input_tokens": recorded.input_tokens, "output_tokens": recorded.output_tokens,
               "blocked": recorded.blocked, "results": results}
    emit("final_result", summary)
    print(json.dumps({k: v for k, v in summary.items() if k != "results"}), flush=True)

def validate_targets(value, sources):
    """Report per-post output defects without withholding another post."""
    if not isinstance(value, dict) or set(value) != {"posts"} or not isinstance(value["posts"], dict) or set(value["posts"]) != set(sources):
        return {slot: ["post_keys_or_root_invalid"] for slot in sources}
    defects = {}
    for slot, row in value["posts"].items():
        errors = []
        if not isinstance(row, dict) or set(row) != {"targets", "other_mentions", "uncertainties"}:
            defects[slot] = ["fields_invalid"]
            continue
        for field, keys in (
            ("targets", {"name", "offering", "evidence", "reason"}),
            ("other_mentions", {"name", "role", "evidence"}),
        ):
            if not isinstance(row[field], list):
                errors.append(f"{field}:not_array")
                continue
            for item in row[field]:
                if not isinstance(item, dict) or set(item) != keys or any(not isinstance(item[k], str) or not item[k].strip() for k in keys):
                    errors.append(f"{field}:item_shape")
                elif item["evidence"] not in sources[slot]["source_text"]:
                    errors.append(f"{field}:evidence_not_verbatim")
        if not isinstance(row["uncertainties"], list) or any(not isinstance(u, str) for u in row["uncertainties"]):
            errors.append("uncertainties:invalid")
        defects[slot] = errors
    return defects


def targets_only(path):
    prior_receipts = json.loads(Path(path).read_text())
    main = prior_receipts["main"]
    diagnostic = prior_receipts["followup"]
    if main["calls"] != 15 or diagnostic["calls"] != 6 or main["blocked"] or diagnostic["blocked"] or diagnostic["task_calls_total"] != 34:
        raise ValueError("unexpected_prior_budget_or_run")
    frozen = target_manifest()
    emit("target_contract", {**frozen, "prior_receipts_sha256": digest(prior_receipts)})
    delegate = build_classifier_client_from_env(load_config(Path("config.yaml")))
    if not isinstance(delegate, DeepInfraChatCompletionsClient) or delegate.model != MODEL or delegate.request_profile != PROFILE:
        raise RuntimeError("configured_direct_classifier_unavailable_or_changed")
    recorded = RecordedTransport(delegate, call_limit=3, cost_limit=Decimal("1"))
    client = replace(delegate, transport=recorded)
    results = []
    for index, cases in enumerate(frozen["batches"], 1):
        sources = source_posts(cases)
        recorded.phase = f"batch{index}:promotion_targets_only"
        response = json_call(client, TARGET_PROMPT, {"posts": sources}, max_tokens=4096)
        result = {
            "batch": index, "case_ids": [c["case_id"] for c in cases],
            "response": response,
            "validation_errors": validate_targets(response.get("value"), sources),
            "promoter_classification": {"status": "not_evaluated", "reason": "author_relationship_out_of_scope"},
        }
        results.append(result)
        emit("batch_result", result)
        if recorded.blocked:
            break
    emit("final_result", {
        "experiment": frozen["experiment"], "manifest_sha256": digest(frozen),
        "calls": recorded.calls, "task_calls_total": 34 + recorded.calls,
        "cost_usd": str(recorded.cost), "input_tokens": recorded.input_tokens,
        "output_tokens": recorded.output_tokens, "blocked": recorded.blocked, "results": results,
    })
